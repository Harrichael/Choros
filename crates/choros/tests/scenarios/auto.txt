# --deps auto lands on only what the commits need. Work that takes back
# what a queued set added needs that set: without it, the removal would
# quietly vanish.

$ printf '[app]\ncheck = "true"\n' > .choros-config/repos.toml
$ choros work one app

mkdir $ROOT/trunk/one

copying agent templates…
$ cd trunk/one/app
$ choros repo bind app appq

$ choros land continue

    landing: $ROOT/trunk/landing/appq/landing-1
$ cd trunk
