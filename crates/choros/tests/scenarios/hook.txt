# A trunk can name a land hook in `.choros-config/hooks.toml`. Each land
# runs it before, when it can change what the land carries or refuse it,
# and after, when it is told what landed.

$ printf '%s\n' 'in=$(cat)' 'echo "hook $1: $in" >&2' 'case "$1 $in" in' '  before*archive*) echo "the tracker says: not finished" >&2; exit 1;;' '  before*) echo "{\"labels\": [\"tasks/a\", \"tasks/b\"], \"priority\": \"p2\"}";;' 'esac' > hook.sh
$ printf 'land = "sh %s/hook.sh"\n' "$PWD" > .choros-config/hooks.toml
$ choros work one app
